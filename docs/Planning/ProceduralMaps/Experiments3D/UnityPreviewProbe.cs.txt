string DirectoryPath = "M:/Prototypes/VaticanDefendor/docs/Planning/ProceduralMaps/Experiments3D/Results";
var Data = Newtonsoft.Json.Linq.JArray.Parse(System.IO.File.ReadAllText(DirectoryPath + "/PreviewData.json"));
var Results = new System.Collections.Generic.List<object>();
var OriginalScene = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
bool OriginalDirty = OriginalScene.isDirty;
var Allocated = new System.Collections.Generic.List<UnityEngine.Object>();
UnityEngine.Mesh Cone(float Radius, float Height, int Sides)
{
    var Vertices = new System.Collections.Generic.List<UnityEngine.Vector3>();
    var Indices = new System.Collections.Generic.List<int>();
    Vertices.Add(new UnityEngine.Vector3(0, Height, 0));
    for (int Side = 0; Side < Sides; Side++)
    {
        float Angle = Side * UnityEngine.Mathf.PI * 2 / Sides;
        Vertices.Add(new UnityEngine.Vector3(UnityEngine.Mathf.Cos(Angle) * Radius, 0, UnityEngine.Mathf.Sin(Angle) * Radius));
    }
    for (int Side = 0; Side < Sides; Side++)
    {
        Indices.Add(0);
        Indices.Add((Side + 1) % Sides + 1);
        Indices.Add(Side + 1);
    }
    var Mesh = new UnityEngine.Mesh();
    Mesh.SetVertices(Vertices);
    Mesh.SetTriangles(Indices, 0);
    Mesh.RecalculateNormals();
    Allocated.Add(Mesh);
    return Mesh;
}
UnityEngine.Mesh Box()
{
    var Vertices = new UnityEngine.Vector3[]
    {
        new UnityEngine.Vector3(-0.5f,-0.5f,-0.5f), new UnityEngine.Vector3(0.5f,-0.5f,-0.5f),
        new UnityEngine.Vector3(0.5f,0.5f,-0.5f), new UnityEngine.Vector3(-0.5f,0.5f,-0.5f),
        new UnityEngine.Vector3(-0.5f,-0.5f,0.5f), new UnityEngine.Vector3(0.5f,-0.5f,0.5f),
        new UnityEngine.Vector3(0.5f,0.5f,0.5f), new UnityEngine.Vector3(-0.5f,0.5f,0.5f)
    };
    int[] Indices = {0,2,1,0,3,2,4,5,6,4,6,7,0,4,7,0,7,3,1,2,6,1,6,5,3,7,6,3,6,2,0,1,5,0,5,4};
    var Mesh = new UnityEngine.Mesh();
    Mesh.vertices = Vertices;
    Mesh.triangles = Indices;
    Mesh.RecalculateNormals();
    Allocated.Add(Mesh);
    return Mesh;
}
UnityEngine.Material Material(UnityEngine.Color Color)
{
    var Result = new UnityEngine.Material(UnityEngine.Shader.Find("Universal Render Pipeline/Lit"));
    Result.SetColor("_BaseColor", Color);
    Result.SetFloat("_Smoothness", 0.15f);
    Result.SetFloat("_Cull", 0);
    Allocated.Add(Result);
    return Result;
}
var Grass = Material(new UnityEngine.Color(0.43f, 0.55f, 0.28f));
var Shore = Material(new UnityEngine.Color(0.65f, 0.58f, 0.37f));
var Rock = Material(new UnityEngine.Color(0.47f, 0.47f, 0.42f));
var Ash = Material(new UnityEngine.Color(0.29f, 0.27f, 0.25f));
var Pine = Material(new UnityEngine.Color(0.15f, 0.29f, 0.18f));
var Trunk = Material(new UnityEngine.Color(0.30f, 0.20f, 0.13f));
var Church = Material(new UnityEngine.Color(0.78f, 0.73f, 0.57f));
var Roof = Material(new UnityEngine.Color(0.57f, 0.16f, 0.11f));
var Water = Material(new UnityEngine.Color(0.16f, 0.44f, 0.51f));
var Lava = Material(new UnityEngine.Color(0.95f, 0.23f, 0.04f));
Lava.EnableKeyword("_EMISSION");
Lava.SetColor("_EmissionColor", new UnityEngine.Color(0.55f, 0.07f, 0.0f));
var BoxMesh = Box();
var TreeMesh = Cone(1.2f, 3.7f, 7);
var TopMesh = Cone(0.95f, 2.8f, 7);
var RoofMesh = Cone(1.0f, 1.6f, 4);
try
{
    foreach (Newtonsoft.Json.Linq.JObject PreviewData in Data)
    {
        if ((int)PreviewData["Seed"] != 105)
        {
            continue;
        }
        string Profile = (string)PreviewData["Id"];
        string Liquid = (string)PreviewData["Liquid"];
        var Preview = new UnityEditor.PreviewRenderUtility();
        try
        {
            Preview.camera.clearFlags = UnityEngine.CameraClearFlags.SolidColor;
            Preview.camera.backgroundColor = new UnityEngine.Color(0.11f, 0.14f, 0.16f);
            Preview.camera.nearClipPlane = 0.1f;
            Preview.camera.farClipPlane = 250;
            Preview.camera.fieldOfView = 39;
            Preview.camera.transform.position = new UnityEngine.Vector3(37, 33, 41);
            Preview.camera.transform.LookAt(UnityEngine.Vector3.zero);
            Preview.camera.allowHDR = false;
            Preview.lights[0].intensity = 1.5f;
            Preview.lights[0].transform.rotation = UnityEngine.Quaternion.Euler(50, -35, 0);
            Preview.lights[1].intensity = 0.7f;
            Preview.ambientColor = new UnityEngine.Color(0.45f, 0.49f, 0.52f);
            var Positions = new System.Collections.Generic.List<UnityEngine.Vector3>();
            foreach (Newtonsoft.Json.Linq.JArray Position in PreviewData["Vertices"])
            {
                Positions.Add(new UnityEngine.Vector3((float)Position[0], (float)Position[1], (float)Position[2]));
            }
            var Indices = new System.Collections.Generic.List<int>();
            foreach (var Index in PreviewData["Triangles"])
            {
                Indices.Add((int)Index);
            }
            var Terrain = new UnityEngine.Mesh();
            Terrain.SetVertices(Positions);
            Terrain.subMeshCount = 2;
            var GroundIndices = new System.Collections.Generic.List<int>();
            var ShoreIndices = new System.Collections.Generic.List<int>();
            for (int Index = 0; Index < Indices.Count; Index += 3)
            {
                var Destination = GroundIndices;
                float Height = (Positions[Indices[Index]].y + Positions[Indices[Index + 1]].y + Positions[Indices[Index + 2]].y) / 3;
                if (Height < 0.55f)
                {
                    Destination = ShoreIndices;
                }
                Destination.Add(Indices[Index]);
                Destination.Add(Indices[Index + 1]);
                Destination.Add(Indices[Index + 2]);
            }
            Terrain.SetTriangles(GroundIndices, 0);
            Terrain.SetTriangles(ShoreIndices, 1);
            Terrain.RecalculateNormals();
            Terrain.RecalculateBounds();
            Allocated.Add(Terrain);
            Preview.BeginPreview(new UnityEngine.Rect(0, 0, 1100, 780), UnityEngine.GUIStyle.none);
            var GroundMaterial = Grass;
            if (Liquid == "lava")
            {
                GroundMaterial = Ash;
            }
            Preview.DrawMesh(Terrain, UnityEngine.Matrix4x4.identity, GroundMaterial, 0);
            Preview.DrawMesh(Terrain, UnityEngine.Matrix4x4.identity, Shore, 1);
            if (Liquid != "none")
            {
                var LiquidVertices = new System.Collections.Generic.List<UnityEngine.Vector3>();
                foreach (var Position in Positions)
                {
                    LiquidVertices.Add(new UnityEngine.Vector3(Position.x, 0.025f, Position.z));
                }
                var LiquidGeometry = new UnityEngine.Mesh();
                LiquidGeometry.SetVertices(LiquidVertices);
                LiquidGeometry.SetTriangles(Indices, 0);
                LiquidGeometry.RecalculateNormals();
                Allocated.Add(LiquidGeometry);
                var LiquidColor = Water;
                if (Liquid == "lava")
                {
                    LiquidColor = Lava;
                }
                Preview.DrawMesh(LiquidGeometry, UnityEngine.Matrix4x4.identity, LiquidColor, 0);
            }
            foreach (Newtonsoft.Json.Linq.JObject ObjectData in PreviewData["Objects"])
            {
                var PositionData = ObjectData["Position"];
                var Position = new UnityEngine.Vector3((float)PositionData[0], (float)PositionData[1], (float)PositionData[2]);
                float Scale = (float)ObjectData["Scale"];
                if ((string)ObjectData["Kind"] == "tree")
                {
                    Preview.DrawMesh(BoxMesh, UnityEngine.Matrix4x4.TRS(Position + UnityEngine.Vector3.up * Scale, UnityEngine.Quaternion.identity, new UnityEngine.Vector3(0.3f, 2, 0.3f) * Scale), Trunk, 0);
                    Preview.DrawMesh(TreeMesh, UnityEngine.Matrix4x4.TRS(Position + UnityEngine.Vector3.up * Scale, UnityEngine.Quaternion.identity, UnityEngine.Vector3.one * Scale), Pine, 0);
                    Preview.DrawMesh(TopMesh, UnityEngine.Matrix4x4.TRS(Position + UnityEngine.Vector3.up * Scale * 2.1f, UnityEngine.Quaternion.identity, UnityEngine.Vector3.one * Scale), Pine, 0);
                }
                else
                {
                    Preview.DrawMesh(BoxMesh, UnityEngine.Matrix4x4.TRS(Position + UnityEngine.Vector3.up * 0.35f, UnityEngine.Quaternion.Euler(15, (float)ObjectData["Rotation"] * 57.3f, 12), new UnityEngine.Vector3(1.4f, 0.9f, 1.1f) * Scale), Rock, 0);
                }
            }
            var ChurchData = PreviewData["Church"];
            var ChurchPosition = new UnityEngine.Vector3((float)ChurchData[0], (float)ChurchData[1], (float)ChurchData[2]);
            Preview.DrawMesh(BoxMesh, UnityEngine.Matrix4x4.TRS(ChurchPosition + UnityEngine.Vector3.up * 1.3f, UnityEngine.Quaternion.identity, new UnityEngine.Vector3(3, 2.6f, 4.2f)), Church, 0);
            Preview.DrawMesh(BoxMesh, UnityEngine.Matrix4x4.TRS(ChurchPosition + UnityEngine.Vector3.up * 2.9f, UnityEngine.Quaternion.Euler(0, 0, 0), new UnityEngine.Vector3(3.5f, 0.5f, 4.7f)), Roof, 0);
            var TowerPosition = ChurchPosition + new UnityEngine.Vector3(0, 2.7f, -1.4f);
            Preview.DrawMesh(BoxMesh, UnityEngine.Matrix4x4.TRS(TowerPosition, UnityEngine.Quaternion.identity, new UnityEngine.Vector3(1.15f, 5.4f, 1.25f)), Church, 0);
            Preview.DrawMesh(RoofMesh, UnityEngine.Matrix4x4.TRS(TowerPosition + UnityEngine.Vector3.up * 2.7f, UnityEngine.Quaternion.Euler(0, 45, 0), UnityEngine.Vector3.one), Roof, 0);
            Preview.Render(true, false);
            var Image = Preview.EndPreview() as UnityEngine.RenderTexture;
            var PreviousTexture = UnityEngine.RenderTexture.active;
            UnityEngine.RenderTexture.active = Image;
            var Capture = new UnityEngine.Texture2D(Image.width, Image.height, UnityEngine.TextureFormat.RGB24, false);
            Capture.ReadPixels(new UnityEngine.Rect(0, 0, Image.width, Image.height), 0, 0);
            var Pixels = Capture.GetPixels();
            for (int PixelIndex = 0; PixelIndex < Pixels.Length; PixelIndex++)
            {
                Pixels[PixelIndex] = Pixels[PixelIndex].gamma;
            }
            Capture.SetPixels(Pixels);
            Capture.Apply();
            UnityEngine.RenderTexture.active = PreviousTexture;
            string ImagePath = DirectoryPath + "/Unity-" + Profile + "-105.png";
            System.IO.File.WriteAllBytes(ImagePath, Capture.EncodeToPNG());
            UnityEngine.Object.DestroyImmediate(Capture);
            Results.Add(new { Profile, VertexCount = Terrain.vertexCount, TriangleCount = Indices.Count / 3, ImagePath, Scope = "Isolated Unity preview rendering; no gameplay or fluid simulation." });
        }
        finally
        {
            Preview.Cleanup();
        }
    }
}
finally
{
    foreach (var Resource in Allocated)
    {
        UnityEngine.Object.DestroyImmediate(Resource);
    }
}
var Report = new { Results, OriginalScene = OriginalScene.path, OriginalDirty, CurrentScene = UnityEngine.SceneManagement.SceneManager.GetActiveScene().path, CurrentDirty = UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty };
System.IO.File.WriteAllText(DirectoryPath + "/UnityPreviewResults.json", Newtonsoft.Json.JsonConvert.SerializeObject(Report, Newtonsoft.Json.Formatting.Indented));
return Report;
